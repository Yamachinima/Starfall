--------------------------------------------------------------------------------
-- 1b. HARDCODED XP PROTECTION  (obfuscated multi-layer encoding)
-- XP rewards are stored in _enc[] as 16-bit position-keyed integers.
-- The decode key for index i uses two independent linear congruences:
--   kl(i) = (i*39 + 95)  mod 256   (low byte)
--   kh(i) = (i*59 + 113) mod 256   (high byte)
-- Decoding: lo = (_enc[i] & 0xFF) XOR kl(i), hi = (_enc[i]>>8) XOR kh(i)
--           XP = sign_extend_16( (hi<<8) | lo )
-- CFG.Tiers carries NO XP data; adding values there is completely ignored.
-- The integrity seal is a position-weighted sum of all 37 tier values plus
-- baseKillXP/maxLevel/maxPrestige; a single-byte edit to _enc silences all rewards.
--------------------------------------------------------------------------------
local getVerifiedTierXP, getVerifiedBaseKillXP, getVerifiedMaxLevel, getVerifiedMaxPrestige
do
    local _bx, _ba, _bo, _bs, _bl, _br, _bt =
        bit.bxor, bit.band, bit.bor, bit.rshift, bit.lshift, bit.arshift, bit.tobit

    local function _kl(i) return _ba(i * 39 + 95,  0xFF) end
    local function _kh(i) return _ba(i * 59 + 113, 0xFF) end

    -- Decode a single entry: XOR bytes with positional keys, then sign-extend 16->32-bit.
    -- _bo (bor) combines the two decoded bytes; _ba (band) was the previous bug here.
    local function _dec(e, i)
        local lo = _bx(_ba(e, 0xFF), _kl(i))
        local hi = _bx(_bs(e, 8),    _kh(i))
        return _br(_bl(_bt(_bo(_bl(hi, 8), lo)), 16), 16)
    end

    -- Encoded XP table.
    -- Indices 1-37: tier XP (order mirrors CFG.Tiers exactly).
    -- Index 38: baseKillXP.  Index 39: maxLevel.  Index 40: maxPrestige.
    -- i=8 encodes Headshot (100 XP).  i=29 encodes Humiliation (-125 XP, sign-extended).
    local _enc = {
        44402, 58691,  8508, 22567, 40946, 55537,  3886, 18931, 33987,
        48800, 64462, 13285, 28711, 43743, 58942,  8501, 23648, 38449,
        54192,  3323, 18536, 33732, 48772, 63843, 13386, 28091, 44012,
        50327, 57161, 23307, 38892, 53975,  2490, 16477, 35696, 46691,
        61894, 13140, 28263, 43388
    }

    -- Decode and cache all values at init time
    local _tierXP = {}
    local _seal   = 0
    for i = 1, 37 do
        _tierXP[i] = _dec(_enc[i], i)
        _seal = _seal + (_tierXP[i] * (i + 7))
    end
    local _bk = _dec(_enc[38], 38)
    local _ml = _dec(_enc[39], 39)
    local _mp = _dec(_enc[40], 40)
    _seal = _seal + (_bk * 3) + (_ml * 13) + (_mp * 17)

    -- Re-verify cached values on every grant (detects in-memory tampering of _tierXP)
    local function _verify()
        local c = 0
        for i = 1, 37 do c = c + (_tierXP[i] * (i + 7)) end
        c = c + (_bk * 3) + (_ml * 13) + (_mp * 17)
        return c == _seal
    end

    getVerifiedTierXP = function(tierIndex)
        if not _verify() then return 0 end
        return _tierXP[tierIndex] or 0
    end

    getVerifiedBaseKillXP = function()
        if not _verify() then return 0 end
        return _bk
    end

    getVerifiedMaxLevel = function()
        if not _verify() then return 55 end
        return _ml
    end

    getVerifiedMaxPrestige = function()
        if not _verify() then return 11 end
        return _mp
    end
end

--------------------------------------------------------------------------------
-- 2. SHARED HELPERS
-- Defined once here so both SERVER and CLIENT use the identical formula.
-- The XP curve is closure-protected: the base and exponent live only here.
--------------------------------------------------------------------------------
local getXpReq
do
    local _base = 3580
    local _exp  = 1.065
    getXpReq = function(lvl)
        return math.floor(_base * (_exp ^ (lvl - 1)))
    end
end

    -- DUPLICATE INSTANCE GUARD

                -- Found a prior instance of this exact chip owned by the same player.
                -- Remove it so the new instance takes over cleanly.
                print(Color(255, 200, 0), "[System] Duplicate chip instance detected. Removing previous instance.")

                break  -- there should only ever be one prior instance; stop after first

        end

    local ACC_SYNC_INTERVAL    = 10   -- sync every N accumulated hits

    local wepKills = {}
    local nemesisKills = {}  -- [entIndex] = number of times that player has killed ply
    local nemesisNames = {}  -- [entIndex] = last known display name for that player
    
    local lastKilledByID = -1
    local quadFeedTracker = {}
    local globalStreaks = {} 
    local victimHistory = {}
    local lastHitgroups = {}
    local oneShotEligible = {}
    local processedCorpses = {}
    local recentDamager = {}
    local firedStickies = {}   -- [inflictor entity] = true; prevents one grenade awarding Stick on multiple kills
    local activeTargets   = {}  -- assist tracker: entity ref -> cumulative damage dealt by ply
    local assistPollActive = false
    
    local rapidKills = 0
    local lastRapidTime = 0
    
    -- Kill Chain tracking: medals earned in the current unbroken rapid chain.
    -- Updated in real-time on each KillChain trigger; sessionBestChain holds the
    -- highest value seen this session and is synced to the client for persistence.
    local currentChainMedals = 0
    local sessionBestChain   = 0
    
    local firstBloodClaimed = false
    local lastSniperTime = 0
    local sniperTickKills = 0
    
    local plyCrouchTime = 0
    local wasCrouching = false

    -- 1 Hammer Unit = 0.75 in; 1 mile = 63,360 in  63,360 / 0.75 = 84,480 HU per mile
    local HU_PER_MILE = 84480

    local EventMedalLookup = {}
    for i, tier in ipairs(CFG.Tiers) do
        if tier.Event then EventMedalLookup[tier.Event] = i - 1 end
    end

    local StreakLookup = {}
    for i, tier in ipairs(CFG.Tiers) do
        if tier.Kills then
            StreakLookup[tier.Kills] = { idx = i - 1, xp = getVerifiedTierXP(i) }
        end
    end

    -- Server-side objectives event map (mirrors client-side EventMap structure).
    -- Built once at init; the server only needs to know which bit to set per kill.
    -- Bit layout matches ObjEvent net message:  kill=1, headshot=2, sniper=4, melee=8, prop=16
    -- (same mask scheme as the reference objectives_system script)
    local b_bor_l = bit.bor

    -- O(1) lookup for rapid-kill medal names. Kills > 8 fall through to "KillChain".
    local RapidKillLookup = {
        [2] = "DoubleKill", [3] = "TripleKill", [4] = "FuryKill",  [5] = "FrenzyKill",
        [6] = "SuperKill",  [7] = "MegaKill",   [8] = "UltraKill"
    }

    -- XP is clamped to a safe 32-bit signed range to prevent overflow at extreme
    -- Master Prestige levels where getXpReq() grows without bound.
    local XP_CLAMP_MAX = 2000000000

    local lastSyncedWepKills = {}  -- snapshot for delta computation; only changed keys are sent
    local lastSyncedNemesis  = {}  -- [playerName] = count; delta sent per sync cycle
    -- Aggregate nemesis kills by name (not entity index) so the delta table
    -- survives entity removal and can be accumulated across sessions on the client.
    local nemesisByName = {}  -- [playerName] = total session kills against ply
    local lastSyncedVictim   = {}  -- [playerName] = count; delta sent per sync cycle
    local victimByName = {}   -- [playerName] = total kills by ply on that player this session
    local sessionBestStreak  = 0   -- highest killstreak reached this session
    local syncPending = false
    local function syncData()
        if syncPending then return end
        syncPending = true
        timer.simple(0.1, function()
            syncPending = false
            -- All player data and extended stats merged into one message to halve
            -- per-kill net overhead (was two separate net.send calls per sync cycle).
            net.start("FullSync")
            net.writeUInt(min_l(pData.totalKills, 4294967295), 32)
            net.writeUInt(min_l(pData.totalDeaths, 4294967295), 32)
            net.writeInt(pData.xp, 32)
            net.writeUInt(pData.lvl, 16)
            net.writeUInt(pData.prestige, 8)
            net.writeUInt(extStats.meleeKills, 32)
            net.writeUInt(extStats.propKills, 32)
            net.writeUInt(extStats.headshots, 32)
            net.writeInt(extStats.accuracyScore, 32)
            net.writeUInt(extStats.accuracyHits, 32)
            net.writeUInt(extStats.accuracyTotal, 32)
            net.writeDouble(extStats.furthestShot)  -- squared HU; double precision needed for large distances
            net.writeFloat(extStats.distance)
            
            local favWep, maxK = "None", 0
            for k, v in pairs(wepKills) do if v > maxK then maxK = v; favWep = k end end
            net.writeString(favWep)
            
            -- Session archNem name kept for backward read-order compatibility;
            -- the client derives the true lifetime nemesis from the delta table below.
            local archNem, maxN = "None", 0
            for id, count in pairs(nemesisKills) do
                if count > maxN then
                    maxN = count
                    archNem = nemesisNames[id] or tostring(id)
                end
            end
            net.writeString(archNem)
            
            -- Kill Chain best, weapon kills delta, and nemesis kills delta
            net.writeUInt(min_l(sessionBestChain, 65535), 16)
            net.writeUInt(min_l(sessionBestStreak, 65535), 16)
            local wepDelta = {}
            for k, v in pairs(wepKills) do
                local prev = lastSyncedWepKills[k] or 0
                if v > prev then wepDelta[k] = v - prev end
            end
            net.writeTable(wepDelta)
            for k, v in pairs(wepKills) do lastSyncedWepKills[k] = v end
            
            local nemDelta = {}
            for name, count in pairs(nemesisByName) do
                local prev = lastSyncedNemesis[name] or 0
                if count > prev then nemDelta[name] = count - prev end
            end
            net.writeTable(nemDelta)
            for name, count in pairs(nemesisByName) do lastSyncedNemesis[name] = count end

            local victimDelta = {}
            for name, count in pairs(victimByName) do
                local prev = lastSyncedVictim[name] or 0
                if count > prev then victimDelta[name] = count - prev end
            end
            net.writeTable(victimDelta)
            for name, count in pairs(victimByName) do lastSyncedVictim[name] = count end
            
            net.send(ply)
        end)
    end

    local function addXP(amount)
        local maxLvl  = getVerifiedMaxLevel()
        local maxPres = getVerifiedMaxPrestige()
        local isMaster = (pData.prestige >= maxPres)
        if pData.lvl >= maxLvl and not isMaster and amount > 0 then return end
        
        pData.xp = pData.xp + amount
        if pData.xp < 0 then pData.xp = 0 end
        -- Clamp to safe 32-bit signed range (Issue #4 fix)
        if pData.xp > XP_CLAMP_MAX then pData.xp = XP_CLAMP_MAX end
        
        local req = getXpReq(pData.lvl)
        local leveledUp = false

        while pData.xp >= req and (pData.lvl < maxLvl or isMaster) do
            pData.xp = pData.xp - req
            pData.lvl = pData.lvl + 1
            req = getXpReq(pData.lvl)
            leveledUp = true
        end

        if pData.lvl >= maxLvl and not isMaster then 
            pData.lvl = maxLvl
            pData.xp = 0 
        end
        
        if leveledUp then
            net.start("LevelUpNotify")
            net.writeUInt(pData.lvl, 16)
            net.send(ply)
        end
        
        syncData()
    end

    local function checkStreak()
        local tierData = StreakLookup[currentStreak]
        
        if not tierData and currentStreak > 30 then
            tierData = StreakLookup[31] 
        end

        if tierData then
            if tierData.xp > 0 then addXP(tierData.xp) end
            net.start("TriggerMedal")
            net.writeUInt(tierData.idx, 8) 
            net.send(ply)
        end
    end

    local function triggerEvent(eventName)
        local idx = EventMedalLookup[eventName]
        if idx then
            local xpReward = getVerifiedTierXP(idx + 1)
            if xpReward ~= 0 then addXP(xpReward) end
            net.start("TriggerMedal")
            net.writeUInt(idx, 8)
            net.send(ply)
        end
    end

    hook.add("PlayerInitialSpawn", "InitData", function(p)
        if p == ply then syncData() end
    end)
    
    -- Checks all current players for any existing frags and marks first blood
    -- claimed if found. Called immediately (mid-session chip loads) and again
    -- after 1 second (catches players whose data arrives late on connect).
    local function checkFirstBloodHistory()
        if firstBloodClaimed then return end
        for _, p in ipairs(find.allPlayers()) do
            if isValid_l(p) and p:getFrags() > 0 then
                firstBloodClaimed = true
                return
            end
        end
    end
    checkFirstBloodHistory()
    timer.simple(1, checkFirstBloodHistory)

    hook.add("EntityRemoved", "CleanupKillstreakData", function(ent)
        if not isValid_l(ent) then return end
        local id = ent:entIndex()
        globalStreaks[id] = nil
        victimHistory[id] = nil
        lastHitgroups[id] = nil
        oneShotEligible[id] = nil
        processedCorpses[id] = nil
        recentDamager[id] = nil
        nemesisKills[id] = nil   -- prevent stale data from recycled entity indices
        nemesisNames[id] = nil
        activeTargets[ent] = nil  -- entity-ref key used by assist tracker
        firedStickies[ent] = nil  -- entity-ref key; clean up spent grenade entries
    end)

    hook.add("scaleplayerdamage", "TrackPlayerHitgroups", function(target, hitgroup, dmginfo)
        if isValid_l(target) then lastHitgroups[target:entIndex()] = hitgroup end
    end)

    -- Shared helper: tags recentDamager if ply (or a projectile they own) damaged target.
    -- Called by both EntityTakeDamage and PostEntityTakeDamage with their explicit params.
    local function tagAsRecentDamager(target, attacker, inflictor)
        if not isValid_l(target) or not target:isPlayer() then return end
        if target == ply then return end  -- never tag the owner as an assisted-suicide target

        local tagged = false
        if isValid_l(attacker) then
            if attacker == ply then tagged = true
            elseif type(attacker.getOwner) == "function" and attacker:getOwner() == ply then tagged = true end
        end
        if not tagged and isValid_l(inflictor) then
            if inflictor == ply then tagged = true
            elseif type(inflictor.getOwner) == "function" and inflictor:getOwner() == ply then tagged = true end
        end

        if tagged then recentDamager[target:entIndex()] = curtime_l() end
    end

    -- EntityTakeDamage fires before damage is applied, so health is still pre-hit here.
    -- This is the correct place to snapshot oneShotEligible (full-health check).
    hook.add("EntityTakeDamage", "CoreDamageTracking", function(target, attacker, inflictor)
        if not isValid_l(target) or not target:isPlayer() then return end
        local hp = target:getHealth()
        if hp >= target:getMaxHealth() and hp > 0 then
            oneShotEligible[target:entIndex()] = curtime_l()
        end
        tagAsRecentDamager(target, attacker, inflictor)
    end)

    -- PostEntityTakeDamage is a backup pass for recentDamager tagging only.
    -- oneShotEligible is deliberately excluded here: health is already reduced.
    hook.add("PostEntityTakeDamage", "BackupDamageTracking", function(target, attacker, inflictor)
        tagAsRecentDamager(target, attacker, inflictor)
    end)

    --------------------------------------------------------------------------------
    -- ASSIST TRACKER
    -- Uses entity references as keys so health-deficit clamping works correctly.
    -- A 0.5 s polling timer shrinks each entry as the target heals; it shuts down
    -- automatically once no targets are actively bleeding.  On death the final
    -- clamped value is compared to maxHP to determine the contribution tier.
    --------------------------------------------------------------------------------
    local function evaluateHeals()
        local activeCount = 0
        for ent, dmg in pairs(activeTargets) do
            if not isValid_l(ent) then
                activeTargets[ent] = nil
            else
                local maxHp = max_l(1, ent:getMaxHealth())
                local curHp = max_l(0, ent:getHealth())
                local deficit = maxHp - curHp
                local newDmg = min_l(dmg, deficit)
                if newDmg <= 0 then
                    activeTargets[ent] = nil
                else
                    activeTargets[ent] = newDmg
                    activeCount = activeCount + 1
                end
            end
        end
        if activeCount == 0 then
            timer.remove("Assist_Eval")
            assistPollActive = false
        end
    end

    hook.add("EntityTakeDamage", "Assist_Track", function(target, attacker, inflictor, damage)
        if attacker ~= ply then return end
        if not isValid_l(target) then return end
        if not target:isPlayer() and not target:isNPC() then return end
        if target == ply then return end
        activeTargets[target] = (activeTargets[target] or 0) + damage
        if not assistPollActive then
            timer.create("Assist_Eval", 0.5, 0, evaluateHeals)
            assistPollActive = true
        end
    end)

    local function checkAssistedSuicide(victim, attacker)
        if not isValid_l(victim) then return end
        local vicID = victim:entIndex()
   
        if not recentDamager[vicID] or (curtime_l() - recentDamager[vicID] > 10) then 
            recentDamager[vicID] = nil 
            return 
        end
    
        local isEnvironment = true
        if isValid_l(attacker) and attacker:isPlayer() and attacker ~= victim then
            isEnvironment = false
        end

        if isEnvironment then triggerEvent("AssistedSuicide") end
        recentDamager[vicID] = nil 
    end
   
    local function evaluateEvents(victim, inflictor, attacker)
        if not isValid_l(victim) then return end
        local vicID = victim:entIndex()
        local ct = curtime_l()
    
        if processedCorpses[vicID] and processedCorpses[vicID] > ct then return end
        processedCorpses[vicID] = ct + 2  -- 2s window guards against duplicate PlayerDeath fires on laggy servers

        local wepClass = ""
        local isSpecialty = false
        
        if isValid_l(inflictor) then
            if inflictor ~= attacker then
                wepClass = inflictor:getClass():lower()
                isSpecialty = not inflictor:isWeapon()
            elseif isValid_l(attacker) and attacker:isPlayer() then
                local w = attacker:getActiveWeapon()
                if isValid_l(w) then wepClass = w:getClass():lower() end
            end
        end

        wepKills[wepClass] = (wepKills[wepClass] or 0) + 1
        currentStreak = currentStreak + 1
        if currentStreak > sessionBestStreak then sessionBestStreak = currentStreak end
        sessionKills = sessionKills + 1
        pData.totalKills = pData.totalKills + 1
        addXP(getVerifiedBaseKillXP())
        checkStreak()
        syncData()

        -- Stick medal: awarded once per grenade detonation when ply kills a player
        -- using a sticky grenade that was physically attached to them at death.
        -- firedStickies prevents the same grenade entity from awarding the medal a
        -- second time if it kills multiple players in the same explosion.
        if isValid_l(inflictor) and inflictor:getClass() == "m9k_thrown_sticky_grenade" then
            local stickyParent = inflictor:getParent()
            if isValid_l(stickyParent) and stickyParent:isPlayer() and not firedStickies[inflictor] then
                firedStickies[inflictor] = true
                triggerEvent("Stick")
            end
        end
        
        if sessionKills == 100 then triggerEvent("GunExpert") end

        -- Rapid kill chain tracking with Kill Chain medal counting.
        -- When the chain breaks (time gap > Rapid_Time), finalize currentChainMedals
        -- and reset for the next chain.
        if ct - lastRapidTime <= CFG.Rapid_Time then
            rapidKills = rapidKills + 1
        else
            -- Chain broke: finalize best chain before resetting
            if currentChainMedals > sessionBestChain then
                sessionBestChain = currentChainMedals
            end
            currentChainMedals = 0
            rapidKills = 1
        end
        lastRapidTime = ct

        local rapidEvent = RapidKillLookup[rapidKills]
        if rapidEvent then
            triggerEvent(rapidEvent)
        elseif rapidKills > 8 then
            triggerEvent("KillChain")
            currentChainMedals = currentChainMedals + 1
            -- Update best in real-time so disconnects don't lose data
            if currentChainMedals > sessionBestChain then
                sessionBestChain = currentChainMedals
            end
        end

        local isHeadshot = (lastHitgroups[vicID] == 1)
        if not isHeadshot and type(victim.lastHitGroup) == "function" then
            isHeadshot = (victim:lastHitGroup() == 1)
        end
        if isHeadshot then extStats.headshots = extStats.headshots + 1; triggerEvent("Headshot") end
    
        if not ply:isAlive() then triggerEvent("Afterlife") end

        if WeaponLookups.Knives[wepClass] then
            extStats.meleeKills = extStats.meleeKills + 1
            triggerEvent("Skewered")
        end
        
        if WeaponLookups.Props[wepClass] then extStats.propKills = extStats.propKills + 1; triggerEvent("Pancake") end
        if globalStreaks[vicID] and globalStreaks[vicID] >= 10 then triggerEvent("Buzzkill") end
        if WeaponLookups.Blades[wepClass] then
            -- Intentional: Blades (machete/damascus) and Knives are mutually exclusive
            -- lookup tables, so a single kill can only reach one of these two branches.
            -- Each independently increments meleeKills because both are melee kills.
            extStats.meleeKills = extStats.meleeKills + 1
            triggerEvent("ChoppedUp")
        end
        if deathStreak >= 3 then triggerEvent("Comeback") end
        if WeaponLookups.Bows[wepClass] then triggerEvent("Deadeye") end
        -- Dropshot: player must be actively crouched at the moment of the kill AND
        -- the kill must land within Dropshot_Window seconds of first crouching.
        -- Cancelling the crouch before the kill forfeits the medal by design --
        -- isCrouching() is an intentional guard, not a latency workaround.
        if ply:isCrouching() and (ct - plyCrouchTime) <= CFG.Dropshot_Window and not isSpecialty then triggerEvent("Dropshot") end

        -- Backstabber covers both knives and blades, matching the medal description.
        if WeaponLookups.Knives[wepClass] or WeaponLookups.Blades[wepClass] then
            local plyAim = ply:getEyeAngles():getForward()
            local vicAim = (victim.getEyeAngles and victim:getEyeAngles():getForward()) or victim:getAngles():getForward()
            -- dot > 0.7 means attacker is within ~45 behind the victim's facing direction
            if plyAim:dot(vicAim) > 0.7 then triggerEvent("Backstabber") end
        end
        
        -- Compute squared distance to avoid sqrt in the kill hot path.
        -- furthestShot is stored as squared HU; math.sqrt is applied at display time only.
        local distSqr = (ply:getPos() - victim:getPos()):getLengthSqr()
        if distSqr > extStats.furthestShot then extStats.furthestShot = distSqr end
        if distSqr > CFG.LongShot_Min_Dist_Sqr and not isSpecialty then triggerEvent("LongShot") end
        
        local isSniper = WeaponLookups.Snipers[wepClass] or false
        
        -- Efficiency: two sniper kills within 0.1s (same bullet passing through multiple targets).
        -- A small time window is more robust than exact tick equality across all server rates.
        if isSniper then
            if (ct - lastSniperTime) <= 0.1 then
                sniperTickKills = sniperTickKills + 1
                if sniperTickKills == 2 then triggerEvent("Efficiency") end
            else
                lastSniperTime = ct
                sniperTickKills = 1
            end
        end
        
        local isOneShot = oneShotEligible[vicID] and (ct - oneShotEligible[vicID]) <= CFG.OneShotOneKill_Window
        if isSniper and isOneShot and not isSpecialty then triggerEvent("OneShotOneKill") end

        local vWep = victim:getActiveWeapon()
        local aWep = isValid_l(attacker) and attacker:getActiveWeapon() or nil
        
        -- Hard validity check first; pcall is only reached if getActiveWeapon() returned invalid.
        if isValid_l(victim) and not isValid_l(vWep) then
            local success, inv = pcall(victim.getWeapons, victim)
            if success and inv and #inv > 0 then vWep = inv[1] end
        end
        
        if isValid_l(vWep) and isValid_l(aWep) then
            if vWep:getClass() == aWep:getClass() then triggerEvent("Backfire") end
        end

        if lastKilledByID == vicID then 
            triggerEvent("Revenge") 
            lastKilledByID = -1 
        end
        
        if ply:isAlive() and ply:getHealth() <= (ply:getMaxHealth() * 0.2) then triggerEvent("Survivor") end
        
        -- Write-pointer compaction: matches the pattern used by activeMedals and lvlNotifs.
        quadFeedTracker[#quadFeedTracker + 1] = ct
        local qfLive = 0
        for i = 1, #quadFeedTracker do
            if ct - quadFeedTracker[i] <= 5 then
                qfLive = qfLive + 1
                quadFeedTracker[qfLive] = quadFeedTracker[i]
            end
        end
        for i = qfLive + 1, #quadFeedTracker do quadFeedTracker[i] = nil end
        
        if qfLive >= 4 then
            triggerEvent("QuadFeed")
            for i = 1, qfLive do quadFeedTracker[i] = nil end
        end

        deathStreak = 0 
        globalStreaks[vicID] = 0 
        lastHitgroups[vicID] = 0
        oneShotEligible[vicID] = nil
        activeTargets[victim] = nil

        -- ObjEvent: send a 12-bit kill-type mask to the client so it can update
        -- objective progress without a second hook or duplicate weapon detection.
        -- Bit layout: kill=1, headshot=2, sniper=4, melee=8, prop=16,
        --             pistol=32, ar=64, shotgun=128, smg=256, lmg=512, explosive=1024
        -- isHeadshot and wepClass are already resolved above; reuse them here.
        do
            local objMask = 1  -- every kill counts as a kill
            if isHeadshot then objMask = b_bor_l(objMask, 2) end
            if WeaponLookups.Snipers[wepClass] and wepClass ~= "weapon_357" then objMask = b_bor_l(objMask, 4) end
            if WeaponLookups.Knives[wepClass] or WeaponLookups.Blades[wepClass]
                                                        then objMask = b_bor_l(objMask, 8)    end
            if WeaponLookups.Props[wepClass]            then objMask = b_bor_l(objMask, 16)   end
            if WeaponLookups.Pistols[wepClass] or wepClass == "weapon_357" then objMask = b_bor_l(objMask, 32) end
            if WeaponLookups.AssaultRifles[wepClass]    then objMask = b_bor_l(objMask, 64)   end
            if WeaponLookups.Shotguns[wepClass]         then objMask = b_bor_l(objMask, 128)  end
            if WeaponLookups.SMGs[wepClass]             then objMask = b_bor_l(objMask, 256)  end
            if WeaponLookups.HeavyAndExplosives[wepClass] and
               not WeaponLookups.Snipers[wepClass]     then objMask = b_bor_l(objMask, 512)  end  -- lmg = m249/m60/minigun/pkp
            if WeaponLookups.HeavyAndExplosives[wepClass] and
               (wepClass:find("rpg") or wepClass:find("rocket") or wepClass:find("grenade") or
                wepClass:find("ied") or wepClass:find("davy") or wepClass:find("m202") or
                wepClass:find("milkor") or wepClass:find("ex41") or wepClass:find("matador") or
                wepClass == "weapon_frag" or wepClass == "weapon_slam")
                                                        then objMask = b_bor_l(objMask, 1024) end
            net.start("ObjEvent")
            net.writeUInt(objMask, 12)
            net.send(ply)
        end
    end

    hook.add("PlayerDeath", "TrackPlayerKills", function(victim, inflictor, attacker)
        local isValidAttacker = isValid_l(attacker)
        local isPlayerAttacker = isValidAttacker and attacker:isPlayer()
        local attID = isValidAttacker and attacker:entIndex() or -1

        -- Clear recentDamager only when a player (not the environment) gets the kill.
        -- Environment deaths are left for checkAssistedSuicide to evaluate and clear.
        if isValid_l(victim) and isPlayerAttacker and attacker ~= victim then
            recentDamager[victim:entIndex()] = nil
        end

        if not firstBloodClaimed and isPlayerAttacker and attacker ~= victim then
            firstBloodClaimed = true
            if attacker == ply then triggerEvent("FirstBlood") end
        end

        if isPlayerAttacker and attacker ~= victim then
            globalStreaks[attID] = (globalStreaks[attID] or 0) + 1
            victimHistory[attID] = victim
            if attacker == ply then
                local vicName = victim:getName()
                victimByName[vicName] = (victimByName[vicName] or 0) + 1
            end
        end

        if attacker ~= ply then
            for i = 1, #quadFeedTracker do quadFeedTracker[i] = nil end
        end

        if victim == ply then
            currentStreak = 0 
            deathStreak = deathStreak + 1
            pData.totalDeaths = pData.totalDeaths + 1
            lastKilledByID = attID
            globalStreaks[ply:entIndex()] = 0
            
            if isPlayerAttacker and attacker ~= ply then
                nemesisKills[attID] = (nemesisKills[attID] or 0) + 1
                -- Name is re-captured on each kill in case of a mid-session rename
                local attName = attacker:getName()
                nemesisNames[attID] = attName
                nemesisByName[attName] = (nemesisByName[attName] or 0) + 1
                local infClass = isValid_l(inflictor) and inflictor:getClass():lower() or ""
                if WeaponLookups.Knives[infClass] or WeaponLookups.Blades[infClass] then triggerEvent("Humiliation") end

                -- DeathCard: send attacker stats to the victim's client.
                local attKills  = math.min(attacker:getFrags(),  65535)
                local attDeaths = math.min(attacker:getDeaths(), 65535)
                local attStreak = math.min(globalStreaks[attID] or 0, 65535)
                local killDist  = (ply:getPos() - attacker:getPos()):getLength()

                -- Resolve the weapon class used to kill ply for the death card display.
                local killWepClass = ""
                if isValid_l(inflictor) and inflictor ~= attacker then
                    killWepClass = inflictor:getClass():lower()
                elseif isValid_l(attacker) and attacker:isPlayer() then
                    local w = attacker:getActiveWeapon()

                net.send(ply)
            end

        elseif attacker == ply and victim ~= ply then
            evaluateEvents(victim, inflictor, attacker)
        else

            -- activeTargets uses entity refs and is health-deficit clamped by evaluateHeals().

                end

    -- then immediately removes itself. Tracking is per-target: switching to a new
    -- target or the target dying resets hits/misses for the current engagement.
    -- The lifetime accuracyScore accumulates hit-minus-miss across all engagements.
    --------------------------------------------------------------------------------
    local function acc_resetTracking()
        if acc_targetEnt then
            acc_targetEnt     = nil
            acc_hits          = 0
            acc_misses        = 0
            acc_hitsSinceSync = 0
        end
    end

        if ent == acc_targetEnt then acc_resetTracking() end
    end)

        if victim == acc_targetEnt then acc_resetTracking() end
    end)

    -- This function only exists in the hook table for a fraction of a millisecond
    local function acc_evaluateTick()
        hook.remove("Tick", "AccTracker_Eval")

        -- Issue #3 fix: use cached isValid_l for consistency with the rest of the codebase
        if not isValid_l(acc_targetEnt) then
            acc_shotsFiredTick = 0
            acc_hitsThisTick   = 0
            return
        end

        -- Stop counting misses if no hit has occurred in the last 5 seconds
        if acc_lastHitTime > 0 and (curtime_l() - acc_lastHitTime) > 5 then
            acc_shotsFiredTick = 0
            acc_hitsThisTick   = 0
            return
        end

        local missedThisTick = acc_shotsFiredTick - acc_hitsThisTick
        if missedThisTick < 0 then missedThisTick = 0 end

        if acc_hitsThisTick > 0 or missedThisTick > 0 then
            acc_hits   = acc_hits   + acc_hitsThisTick
            acc_misses = acc_misses + missedThisTick
            -- Each hit adds 1 to lifetime score, each miss subtracts 1
            extStats.accuracyScore = extStats.accuracyScore + acc_hitsThisTick - missedThisTick
            extStats.accuracyHits  = extStats.accuracyHits  + acc_hitsThisTick
            extStats.accuracyTotal = extStats.accuracyTotal + acc_hitsThisTick + missedThisTick
            -- Only push a FullSync after ACC_SYNC_INTERVAL accumulated hits to avoid
            -- spamming the network during sustained firefights. Kill/death paths call
            -- syncData() unconditionally, so data is never lost on engagement end.
            acc_hitsSinceSync = acc_hitsSinceSync + acc_hitsThisTick
            if acc_hitsSinceSync >= ACC_SYNC_INTERVAL then
                acc_hitsSinceSync = 0
                syncData()
            end
        end

        acc_shotsFiredTick = 0
        acc_hitsThisTick   = 0
    end

    local function djb2_hash(str)
        local hash = 5381
        for i = 1, #str do 
            hash = b_xor(bit.tobit(bit.lshift(hash, 5) + hash), s_byte(str, i)) 
        end
        return hash
    end

    local function encryptData(payloadTable)
        local rawData = j_encode(payloadTable)
        local salt = s_fmt("%04x", m_rand(0, 65535))
        local dynamicKey = SAVE_KEY_PREFIX .. salt
        local keyLen = #dynamicKey

        local res = { salt }
        local prevBlock = 0x5A 
        local keyIdx = 1

        for i = 1, #rawData do
            local currentByte = s_byte(rawData, i)
            local keyByte = s_byte(dynamicKey, keyIdx)
            
            keyIdx = keyIdx + 1
            if keyIdx > keyLen then keyIdx = 1 end

            local cipherByte = b_xor(currentByte, keyByte, prevBlock)
            res[#res + 1] = s_char(cipherByte)
            prevBlock = cipherByte
        end
        
        return t_cat(res)
    end

    local function decryptData(binaryData)
        local salt = s_sub(binaryData, 1, 4)
        local dynamicKey = SAVE_KEY_PREFIX .. salt
        local keyLen = #dynamicKey

        local res = {}
        local prevBlock = 0x5A
        local keyIdx = 1

        for i = 5, #binaryData do
            local currentByte = s_byte(binaryData, i)
            
            local keyByte = s_byte(dynamicKey, keyIdx)
            keyIdx = keyIdx + 1
            if keyIdx > keyLen then keyIdx = 1 end
            
            local plainByte = b_xor(currentByte, keyByte, prevBlock)
            res[#res + 1] = s_char(plainByte)
            prevBlock = currentByte
        end
        return j_decode(t_cat(res))
    end

    -- pickCycle: randomly selects 'count' objectives from pool, assigns random
    -- req and proportional xp.
    -- 'seed' is always an explicit integer supplied by loadObjState -- either
    -- a freshly generated os.time() value (on genuine rollover) or the value
    -- restored from disk (on reload).  math.randomseed is called here and only
    -- here, scoped to this function, so chip reloads replay the identical shuffle.
    local function pickCycle(pool, count, reqMin, reqMax, xpMin, xpMax, seed)
        -- StarfallEx does not expose math.randomseed in all builds.
        -- Warm the RNG manually: XOR-mix the seed into several math.random calls
        -- so the shuffle is deterministic from the seed without randomseed.
        -- We call math.random(seed % 2147483647 + 1) twice to advance internal state.
        local s = (seed % 2147483647) + 1
        m_rand(s); m_rand(s)
        -- Shuffle a copy of the pool indices so we don't repeat objectives
        local n   = #pool
        local idx = {}
        for i = 1, n do idx[i] = i end
        for i = n, 2, -1 do
            local j = math.random(1, i)
            idx[i], idx[j] = idx[j], idx[i]
        end
        local result = {}
        for i = 1, math.min(count, n) do
            local tmpl = pool[idx[i]]
            local req  = math.random(reqMin, reqMax)
            -- Round to nearest 5 (daily) or 10 (weekly) for clean numbers
            local snap = (reqMax > 200) and 10 or 5
            req = math.max(reqMin, math.floor((req + snap / 2) / snap) * snap)
            -- XP scales linearly with req across the defined range
            local frac = (req - reqMin) / math.max(reqMax - reqMin, 1)
            local xp   = math.floor(xpMin + frac * (xpMax - xpMin))
            result[#result + 1] = { id = tmpl.id, req = req, xp = xp }
        end
        return result
    end

    local function forceSaveStats()
        local innerData = j_encode({ pStats = pStats, medalStats = medalStats, clStats = clStats, favWepRecord = favWepRecord, archNemRecord = archNemRecord, topVictimRecord = topVictimRecord, activeKeyBind = activeKeyBind, objState = ObjState })
        local checksum = djb2_hash(innerData)
        local securePayload = { data = innerData, hash = checksum }
        
        local encryptedBlock = encryptData(securePayload)

        local success, err = pcall(f_write, TEMP_FILE, encryptedBlock)

        if not success then
            saveFailureCount = saveFailureCount + 1
            -- Exponential backoff: 30s, 60s, 120s, then cap at 300s (5 min).
            -- The normal 5s StatsDiskSaveTimer is paused during backoff by removing it;
            -- a one-shot timer reschedules it once the backoff window expires.
            local backoff = math.min(30 * (2 ^ (saveFailureCount - 1)), 300)
            timer.remove("StatsDiskSaveTimer")
            print(Color(255, 80, 0), s_fmt("[System] Disk write failed (temp)  attempt %d. Retrying in %ds.", saveFailureCount, backoff))
            timer.simple(backoff, function()
                -- Restart the periodic save timer; statsNeedSave will be true so
                -- the next tick immediately retries the write.
                statsNeedSave = true
                timer.create("StatsDiskSaveTimer", 5, 0, function()
                    if statsNeedSave then
                        forceSaveStats()
                        statsNeedSave = false
                    end
                end)
            end)
            return
        end

        local ok2, err2 = pcall(f_write, STATS_FILE, encryptedBlock)
        if not ok2 then
            -- Temp file survives as a recoverable backup for the next load cycle.
            print(Color(255, 100, 0), "[System] Disk write failed (main). Temp backup preserved.")
            return
        end

        -- Successful save: reset failure counter and clean up the temp file.
        saveFailureCount = 0
        f_delete(TEMP_FILE)
    end

    local function attemptLoad(targetFile)
        if not f_exists(targetFile) then return false end

        local rawHex = f_read(targetFile)
        local success, parsed = pcall(decryptData, rawHex)

        if success and type(parsed) == "table" and parsed.data and parsed.hash then
            if djb2_hash(parsed.data) == parsed.hash then
                local decodedData = j_decode(parsed.data)
                if type(decodedData) == "table" then
                    pStats = decodedData.pStats or pStats
                    medalStats = decodedData.medalStats or medalStats

                    if decodedData.clStats then
                        for k, v in pairs(decodedData.clStats) do
                            clStats[k] = v
                        end
                    end
                    -- Ensure armory sub-table is always a table, even on pre-armory saves
                    if type(clStats.armory) ~= "table" then clStats.armory = {} end
                    if type(clStats.armoryFavorites) ~= "table" then clStats.armoryFavorites = {} end
                    if type(clStats.armoryMats) ~= "table" then clStats.armoryMats = {} end
                    
                    -- Restore champion records and keybind
                    if type(decodedData.favWepRecord) == "table" then
                        favWepRecord = decodedData.favWepRecord
                    end
                    if type(decodedData.archNemRecord) == "table" then
                        archNemRecord = decodedData.archNemRecord
                    end
                    if type(decodedData.topVictimRecord) == "table" then
                        topVictimRecord = decodedData.topVictimRecord
                    end
                    if type(decodedData.activeKeyBind) == "number" then
                        activeKeyBind = decodedData.activeKeyBind
                    end
                    -- Restore ObjState (merged into main save for tamper protection)
                    if type(decodedData.objState) == "table" then
                        if type(decodedData.objState.Progress) == "table" then
                            ObjState = decodedData.objState
                        end
                    end
                    -- Sync clStats display fields from champion records
                    clStats.favWep    = favWepRecord.name
                    clStats.archNem   = archNemRecord.name
                    clStats.topVictim = topVictimRecord.name
                    rebuildArmoryList()
                    
                    return true
                end
            else
                print(Color(255, 100, 0), "[Security] Payload checksum mismatch in " .. targetFile .. ". Tampering detected.")
            end
        end

        return false
    end

    -- loadStats() after the main payload is decoded into ObjState.

    local function loadObjState()
        local curDay, curWeek = getTimeIDs()
        local changed = false

        -- XP_VER: bump this number any time the XP ranges change.
        -- If the stored version doesn't match, all objectives are re-rolled
        -- immediately so the new XP values take effect without waiting for
        -- the next natural daily/weekly reset.
        local XP_VER = 3

        -- Force re-roll if XP version is stale (covers the tripled-XP update)
        if (ObjState.xpVer or 0) < XP_VER then
            for k in pairs(ObjState.Progress) do ObjState.Progress[k] = nil end
            local daySeed  = os_time()
            local weekSeed = os_time() + 1  -- +1 avoids identical seeds when called within same second
            ObjState.DaySeed  = daySeed
            ObjState.WeekSeed = weekSeed
            ObjState.ActiveDaily = pickCycle(
                ObjectivePool.Daily, DAILY_COUNT,
                DAILY_REQ_MIN, DAILY_REQ_MAX, DAILY_XP_MIN, DAILY_XP_MAX, daySeed)
            ObjState.ActiveWeekly = pickCycle(
                ObjectivePool.Weekly, WEEKLY_COUNT,
                WEEKLY_REQ_MIN, WEEKLY_REQ_MAX, WEEKLY_XP_MIN, WEEKLY_XP_MAX, weekSeed)
            ObjState.DayID   = curDay
            ObjState.WeekID  = curWeek
            ObjState.xpVer   = XP_VER
            buildActiveObjectives()
            saveObjState()
            return
        end

        -- Roll a new daily cycle when the day changes
        if ObjState.DayID ~= curDay then
            for k in pairs(ObjState.Progress) do
                if string.sub(k, 1, 2) == "d_" then ObjState.Progress[k] = nil end
            end
            local daySeed = os_time()
            ObjState.DaySeed = daySeed
            ObjState.ActiveDaily = pickCycle(
                ObjectivePool.Daily, DAILY_COUNT,
                DAILY_REQ_MIN, DAILY_REQ_MAX, DAILY_XP_MIN, DAILY_XP_MAX, daySeed)
            ObjState.DayID = curDay
            changed = true
        end

        -- Roll a new weekly cycle when the week changes
        if ObjState.WeekID ~= curWeek then
            for k in pairs(ObjState.Progress) do
                if string.sub(k, 1, 2) == "w_" then ObjState.Progress[k] = nil end
            end
            local weekSeed = os_time() + 1  -- +1 avoids identical seeds if day and week roll simultaneously
            ObjState.WeekSeed = weekSeed
            ObjState.ActiveWeekly = pickCycle(
                ObjectivePool.Weekly, WEEKLY_COUNT,
                WEEKLY_REQ_MIN, WEEKLY_REQ_MAX, WEEKLY_XP_MIN, WEEKLY_XP_MAX, weekSeed)
            ObjState.WeekID = curWeek
            changed = true
        end

        -- Stamp IDs on a totally fresh save (DaySeed/WeekSeed = 0 on a brand-new save)
        if ObjState.DayID == "" then
            local daySeed = os_time()
            ObjState.DaySeed = daySeed
            ObjState.ActiveDaily = pickCycle(
                ObjectivePool.Daily, DAILY_COUNT,
                DAILY_REQ_MIN, DAILY_REQ_MAX, DAILY_XP_MIN, DAILY_XP_MAX, daySeed)
            ObjState.DayID = curDay; changed = true
        end
        if ObjState.WeekID == "" then
            local weekSeed = os_time() + 1
            ObjState.WeekSeed = weekSeed
            ObjState.ActiveWeekly = pickCycle(
                ObjectivePool.Weekly, WEEKLY_COUNT,
                WEEKLY_REQ_MIN, WEEKLY_REQ_MAX, WEEKLY_XP_MIN, WEEKLY_XP_MAX, weekSeed)
            ObjState.WeekID = curWeek; changed = true
        end

        -- Ensure Active arrays exist on saves predating the cycle system.
        -- Seeds may be 0 here (pre-fix save); os.time() is used to generate them
        -- once, then stored so subsequent reloads replay the same result.
        if not ObjState.ActiveDaily or #ObjState.ActiveDaily == 0 then
            local daySeed = (ObjState.DaySeed ~= 0) and ObjState.DaySeed or os_time()
            ObjState.DaySeed = daySeed
            ObjState.ActiveDaily = pickCycle(
                ObjectivePool.Daily, DAILY_COUNT,
                DAILY_REQ_MIN, DAILY_REQ_MAX, DAILY_XP_MIN, DAILY_XP_MAX, daySeed)
            changed = true
        end
        if not ObjState.ActiveWeekly or #ObjState.ActiveWeekly == 0 then
            local weekSeed = (ObjState.WeekSeed ~= 0) and ObjState.WeekSeed or (os_time() + 1)
            ObjState.WeekSeed = weekSeed
            ObjState.ActiveWeekly = pickCycle(
                ObjectivePool.Weekly, WEEKLY_COUNT,
                WEEKLY_REQ_MIN, WEEKLY_REQ_MAX, WEEKLY_XP_MIN, WEEKLY_XP_MAX, weekSeed)
            changed = true
        end

        -- Build the live ObjectivesList and ObjEventMap from the active sets
        buildActiveObjectives()

        if changed then saveObjState() end
    end

    local function loadStats()
        -- Safety assertion: SAVE_KEY_PREFIX must be set by InitIO_SteamValidation
        -- before any file IO or net send occurs.  If this fires, it means the call
        -- path was changed and now bypasses Steam ID validation -- fix the caller.
        assert(SAVE_KEY_PREFIX ~= nil and SAVE_KEY_PREFIX ~= "",
            "[System] loadStats() called before SAVE_KEY_PREFIX was initialised. Aborting to prevent unkeyed IO.")

        local loaded = attemptLoad(STATS_FILE)
        
        if not loaded and f_exists(TEMP_FILE) then
            print(Color(255, 200, 0), "[System] Main savestate failed validation, Falling back to shadow backup...")
            loaded = attemptLoad(TEMP_FILE)
        end

        if loaded then
            print(Color(0, 200, 0), "[System] Savestate Loaded successfully!")
        else
            if f_exists(STATS_FILE) or f_exists(TEMP_FILE) then
                local corruptName = "yamastatistics_corrupted_" .. os_time() .. ".txt"
                if f_exists(STATS_FILE) then f_write(corruptName, f_read(STATS_FILE)) end
                f_delete(STATS_FILE)
                f_delete(TEMP_FILE)
                print(Color(255, 50, 50), "[System] Critical decryption failure. Savestate wiped and quarantined: " .. corruptName)
                statsNeedSave = true 
            end
        end

        -- Run cycle reset logic (daily/weekly reset, ID stamping) on the now-loaded ObjState.
        loadObjState()

        initStats()

        -- Restore saved skin selections for favourited weapons
        if type(clStats.armoryMats) == "table" then
            for wep, mat in pairs(clStats.armoryMats) do
                if clStats.armoryFavorites[wep] and mat ~= "" then
                    armoryMats[wep] = mat
                end
            end
        end
        net.start("RestoreStats")
        net.writeInt(pStats.xp, 32)
        net.writeUInt(pStats.lvl, 16)
        net.writeUInt(pStats.prestige, 8)
        net.send()
        
        timer.create("StatsDiskSaveTimer", 5, 0, function()
            if statsNeedSave then 
                forceSaveStats() 
                statsNeedSave = false 
            end
        end)
    end

    -- There is no hardcoded salt in the script: the key material comes entirely
    -- from the player's own account, so there is nothing static here to extract.
    -- The derivation method is visible in the script, but an attacker still needs

    -- Weapon XP curve: 1000 * 1.04^(lvl-1).  Evaluated per-frame only for visible rows
    -- so no pre-bake needed; the formula is O(1) and the result fits in a double.
    local function getWeaponXpReq(lvl)
        return m_floor(1000 * (1.04 ^ (lvl - 1)))

    -- Credit XP and kills to a weapon entry in clStats.armory.
    -- Creates the entry on first use.  Handles multi-level overflow correctly
    -- (a single large XP grant can advance past more than one level).
    -- Returns true if the weapon levelled up (used to mark RT dirty).
    local function armoryAddXP(wepClass, xpAmount, killCredit)
        if not wepClass or wepClass == "" then return false end
        local entry = clStats.armory[wepClass]
        if not entry then
            clStats.armory[wepClass] = { lvl = 1, xp = 0, kills = 0 }
            entry = clStats.armory[wepClass]
        end
        if killCredit then entry.kills = entry.kills + killCredit end
        entry.xp = entry.xp + xpAmount
        local leveledUp = false
        local req = getWeaponXpReq(entry.lvl)
        while entry.xp >= req do
            entry.xp  = entry.xp - req
            entry.lvl = entry.lvl + 1
            req = getWeaponXpReq(entry.lvl)
            leveledUp = true
        end
        return leveledUp
    end

    -- Single receiver for the merged FullSync message (was two separate messages).
    -- Read order must exactly match the server write order in syncData().
    local lastSessionStats = { meleeKills = 0, propKills = 0, headshots = 0, accuracyScore = 0, accuracyHits = 0, accuracyTotal = 0, distance = 0 }

    net.receive("FullSync", function()
        -- Player data block
        local newKills = net.readUInt(32)
        local newDeaths = net.readUInt(32)
        local dKills  = newKills  - pStats.kills
        local dDeaths = newDeaths - pStats.deaths
        if dKills  > 0 then pStats.lifetimeKills  = pStats.lifetimeKills  + dKills  end
        if dDeaths > 0 then pStats.lifetimeDeaths = pStats.lifetimeDeaths + dDeaths end
        pStats.kills   = newKills
        pStats.deaths  = newDeaths
        pStats.xp      = net.readInt(32)
        pStats.lvl     = net.readUInt(16)
        pStats.prestige = net.readUInt(8)

        -- Extended stats block
        local sMelee    = net.readUInt(32)
        local sProp     = net.readUInt(32)
        local sHead     = net.readUInt(32)
        local sAccScore = net.readInt(32)
        local sAccHits  = net.readUInt(32)
        local sAccTotal = net.readUInt(32)
        local sFurthest = net.readDouble()   -- squared HU; double matches server writeDouble
        local sDist     = net.readFloat()
        local sFavWep   = net.readString()    -- session favorite (kept for archNem read order)
        local sArchNem  = net.readString()
        
        -- New fields: kill chain best, weapon kills delta, nemesis kills delta, victim kills delta
        local sBestChain = net.readUInt(16)
        local sBestStreak = net.readUInt(16)
        local sWepDelta  = {}
        local readOk, readResult = pcall(net.readTable)
        if readOk and type(readResult) == "table" then sWepDelta = readResult end
        local sNemDelta  = {}
        local readOk2, readResult2 = pcall(net.readTable)
        if readOk2 and type(readResult2) == "table" then sNemDelta = readResult2 end
        local sVictimDelta = {}
        local readOk3, readResult3 = pcall(net.readTable)
        if readOk3 and type(readResult3) == "table" then sVictimDelta = readResult3 end

        clStats.meleeKills    = clStats.meleeKills + m_max(0, sMelee - lastSessionStats.meleeKills)
        clStats.propKills     = clStats.propKills  + m_max(0, sProp  - lastSessionStats.propKills)
        clStats.headshots     = clStats.headshots  + m_max(0, sHead  - lastSessionStats.headshots)
        -- accuracyScore is hit-minus-miss and can theoretically go negative under sustained
        -- missed fire; clamp to 0 so the saved/accumulated value is always non-negative.
        clStats.accuracyScore = m_max(0, clStats.accuracyScore + (sAccScore - lastSessionStats.accuracyScore))
        clStats.accuracyHits  = clStats.accuracyHits  + m_max(0, sAccHits  - lastSessionStats.accuracyHits)
        clStats.accuracyTotal = clStats.accuracyTotal + m_max(0, sAccTotal - lastSessionStats.accuracyTotal)
        clStats.distance      = clStats.distance   + m_max(0, sDist  - lastSessionStats.distance)
        if sFurthest > clStats.furthestShot then clStats.furthestShot = sFurthest end

        lastSessionStats.meleeKills    = sMelee
        lastSessionStats.propKills     = sProp
        lastSessionStats.headshots     = sHead
        lastSessionStats.accuracyScore = sAccScore
        lastSessionStats.accuracyHits  = sAccHits
        lastSessionStats.accuracyTotal = sAccTotal
        lastSessionStats.distance      = sDist

        -- Biggest Kill Chain: persist the all-time best across sessions
        if sBestChain > clStats.biggestKillChain then
            clStats.biggestKillChain = sBestChain
        end

        -- Highest Killstreak: persist the all-time peak streak across sessions
        if sBestStreak > clStats.highestStreak then
            clStats.highestStreak = sBestStreak
        end

        -- Favorite Weapon: accumulate each sync's delta into the session running total,
        -- then compare the session leader's full total against the saved lifetime champion.
        -- Using the full session total (not the per-sync delta) ensures 200 AK kills
        -- across many syncs correctly beats a saved record of 80 arc kills.
        for wep, delta in pairs(sWepDelta) do
            if delta > 0 then
                sessionWepCounts[wep] = (sessionWepCounts[wep] or 0) + delta
                -- Armory: credit base kill XP per kill delta for this weapon.
                -- 125 XP per kill mirrors CFG.Leveling.Base_Kill_XP (the same value
                -- used server-side); it is hardcoded here so armory progression does
                -- not require an extra net field.
                local leveled = armoryAddXP(wep, delta * 125, delta)
                if leveled and UI.tab == TAB_ARMORY then UI.needsUpdate = true end
            end
        end
        -- Find the session leader by scanning the (small, session-only) table
        local sessWepName, sessWepCount = "None", 0
        for wep, count in pairs(sessionWepCounts) do
            if count > sessWepCount then sessWepCount = count; sessWepName = wep end
        end
        -- Replace the lifetime champion if the session leader beats it
        if sessWepName ~= "None" then
            if sessWepName == favWepRecord.name then
                -- Same weapon: update count to the authoritative session total
                favWepRecord.count = m_max(favWepRecord.count, sessWepCount)
            elseif sessWepCount > favWepRecord.count then
                favWepRecord.name  = sessWepName
                favWepRecord.count = sessWepCount
            end
        end
        clStats.favWep = favWepRecord.name

        -- Arch Nemesis: same accumulator pattern as Favorite Weapon.
        for name, delta in pairs(sNemDelta) do
            if delta > 0 then
                sessionNemCounts[name] = (sessionNemCounts[name] or 0) + delta
            end
        end
        local sessNemName, sessNemCount = "None", 0
        for name, count in pairs(sessionNemCounts) do
            if count > sessNemCount then sessNemCount = count; sessNemName = name end
        end
        if sessNemName ~= "None" then
            if sessNemName == archNemRecord.name then
                archNemRecord.count = m_max(archNemRecord.count, sessNemCount)
            elseif sessNemCount > archNemRecord.count then
                archNemRecord.name  = sessNemName
                archNemRecord.count = sessNemCount
            end
        end
        clStats.archNem = archNemRecord.name

        -- Top Victim: same accumulator pattern as Arch Nemesis, but tracks who ply kills most.
        for name, delta in pairs(sVictimDelta) do
            if delta > 0 then
                sessionVictimCounts[name] = (sessionVictimCounts[name] or 0) + delta
            end
        end
        local sessVicName, sessVicCount = "None", 0
        for name, count in pairs(sessionVictimCounts) do
            if count > sessVicCount then sessVicCount = count; sessVicName = name end
        end
        if sessVicName ~= "None" then
            if sessVicName == topVictimRecord.name then
                topVictimRecord.count = m_max(topVictimRecord.count, sessVicCount)
            elseif sessVicCount > topVictimRecord.count then
                topVictimRecord.name  = sessVicName
                topVictimRecord.count = sessVicCount
            end
        end
        clStats.topVictim = topVictimRecord.name

        updateUICache()
        if UI.tab == TAB_STAT then UI.needsUpdate = true end
        queueSaveStats()
    end)

    -- ObjEvent: 12-bit kill-type mask from server.
    -- Dispatch increments progress for every matching objective that is not yet complete.
    -- On first-time completion a notification is queued and ObjState is saved to disk.
    net.receive("ObjEvent", function()
        local b_band_l = bit.band
        local mask = net.readUInt(12)
        local ct   = curtime_l()

        local function processObjType(typeStr)
            local list = ObjEventMap[typeStr]
            for i = 1, #list do
                local obj  = list[i]
                local cur  = ObjState.Progress[obj.id] or 0
                if cur < obj.req then
                    ObjState.Progress[obj.id] = cur + 1
                    -- On completion: queue HUD notification and grant XP bonus
                    if ObjState.Progress[obj.id] >= obj.req then
                        ObjCompletions[#ObjCompletions + 1] = {
                            id        = obj.id,
                            title     = obj.isDaily and "DAILY OBJECTIVE COMPLETE" or "WEEKLY OBJECTIVE COMPLETE",
                            name      = obj.name,
                            startTime = ct,
                            endTime   = ct + CFG.Obj_Notify_Time
                        }
                        if obj.isDaily then
                            clStats.dailyCompleted = (clStats.dailyCompleted or 0) + 1
                        else
                            clStats.weeklyCompleted = (clStats.weeklyCompleted or 0) + 1
                        end
                        Cached_TotalObjectives = (clStats.dailyCompleted or 0) + (clStats.weeklyCompleted or 0)
                        queueSaveStats()
                        if #lvlNotifs < LVL_NOTIF_MAX then
                            lvlNotifs[#lvlNotifs + 1] = {
                                startTime  = ct + 0.2,
                                displayStr = "OBJECTIVE: +" .. obj.xp .. " XP"
                            }
                        end
                    end
                    saveObjState()
                    if UI.tab == TAB_OBJ then UI.needsUpdate = true end
                end
            end
        end

        if b_band_l(mask,    1) > 0 then processObjType("kill")      end
        if b_band_l(mask,    2) > 0 then
            -- Armory: headshot bonus XP (50) credited to the currently held weapon
            -- on every headshot, independent of objective progress.
            local hsWep = player():getActiveWeapon()
            if isValid(hsWep) then
                local leveled = armoryAddXP(hsWep:getClass():lower(), 50, 0)
                if leveled and UI.tab == TAB_ARMORY then UI.needsUpdate = true end
            end
            processObjType("headshot")
        end
        if b_band_l(mask,    4) > 0 then processObjType("sniper")     end
        if b_band_l(mask,    8) > 0 then processObjType("melee")      end
        if b_band_l(mask,   16) > 0 then processObjType("prop")       end
        if b_band_l(mask,   32) > 0 then processObjType("pistol")     end
        if b_band_l(mask,   64) > 0 then processObjType("ar")         end
        if b_band_l(mask,  128) > 0 then processObjType("shotgun")    end
        if b_band_l(mask,  256) > 0 then processObjType("smg")        end
        if b_band_l(mask,  512) > 0 then processObjType("lmg")        end
        if b_band_l(mask, 1024) > 0 then processObjType("explosive")  end
    end)

    -- Scroll drag think: registered only while the user is actively dragging the
    -- scrollbar thumb. Zero per-frame cost when the panel is closed or idle.
    -- Must be declared before inputPressed so the closure captures this upvalue.
    local function scrollDragThink()
        local g = getPanelGeom(Cached_HUD_SW, Cached_HUD_SH)
        local trackH = g.rowEndY - g.rowStartY
        local totalRowH
        if UI.tab == TAB_ARMORY then
            totalRowH = ARMORY_TOTAL_H
        else
            totalRowH = #MEDAL_LIST * CFG.Stats_Row_H
        end
        local thumbH    = m_max(30, trackH * (g.listAreaH / m_max(totalRowH, 1)))
        local scrollRange = trackH - thumbH
        if scrollRange <= 0 then return end
        local _, my = input.getCursorPos()
        local delta = my - UI.dragStartY
        local scrollDelta = (delta / scrollRange) * UI.maxScroll
        UI.scroll = m_max(0, m_min(UI.maxScroll, UI.dragStartScroll + scrollDelta))
        UI.scrollTarget = UI.scroll
    end

    -- Keys excluded from keybind capture. Mouse buttons 1-5 are blocked to prevent
    -- accidentally binding the stats panel to a combat mouse button while the prompt
    -- is open. ESCAPE (70) cancels capture. Mouse1 (107) is the capture trigger itself.
    -- KEY CODES: mouse1=107, mouse2=108, mouse3=109, mouse4=117, mouse5=118
    local KEYBIND_EXCLUDED = { [70]=true, [107]=true, [108]=true, [109]=true, [117]=true, [118]=true }

    hook.add("InputPressed", "UIInput", function(k)
        -- Keybind capture mode: accept any non-excluded key as the new menu keybind.
        -- ESCAPE (70) cancels capture without changing the bind.
        -- Mouse buttons 1-5 are silently ignored to prevent accidental combat-key binds.
        if UI and UI.waitingForKeybind then
            if not KEYBIND_EXCLUDED[k] then
                activeKeyBind = k
                UI.waitingForKeybind = false
                UI.needsUpdate = true
                queueSaveStats()
            elseif k == 70 then
                -- ESCAPE: cancel capture, no change
                UI.waitingForKeybind = false
                UI.needsUpdate = true
            end
            return
        end

        if k == activeKeyBind then
            if not UI.open then openPanel() else closePanel() end
            return
        end

        if not UI.open then return end

        local g = getPanelGeom(Cached_HUD_SW, Cached_HUD_SH)

        local scrollStep = (UI.tab == TAB_ARMORY) and 75 or (CFG.Stats_Row_H * 3)
        if k == 201 then UI.scroll = m_max(0, UI.scroll - scrollStep) return end
        if k == 202 then UI.scroll = m_min(UI.maxScroll, UI.scroll + scrollStep) return end
        
        if k == 107 then
            local mx, my = input.getCursorPos()

            if hit(UI.btnClose.x, UI.btnClose.y, UI.btnClose.w, UI.btnClose.h, mx, my) then
                closePanel()
                return
            end
            
            -- Keybind box click: enter capture mode

                UI.waitingForKeybind = true
                UI.needsUpdate = true

            end

                    net.start("DoPrestige")
                    net.send()

                local trackX, trackW, trackY, trackH = g.px + g.pw - 10, 8, g.rowStartY, g.rowEndY - g.rowStartY

                    if rowY + CFG.Stats_Row_H > g.rowStartY and rowY < g.rowEndY then

                            UI.selectedMedal = (UI.selectedMedal == entry.id) and nil or entry.id

                        end
                    end
                end

                local rowH  = 32

                local menuH = titleH + #ARMORY_MATERIALS * rowH + 8

                if hit(menuX, menuY, menuW, menuH, mx, my) then
                    -- Determine which option row was clicked
                    for i, matEntry in ipairs(ARMORY_MATERIALS) do
                        local optY = menuY + titleH + (i - 1) * rowH
                        if hit(menuX, optY, menuW, rowH, mx, my) then
                            -- Check weapon level gate before allowing selection
                            local ctxEntry = clStats.armory[ctx.wepName]
                            local ctxLvl   = ctxEntry and ctxEntry.lvl or 1
                            if ctxLvl < matEntry.unlockLvl then
                                -- Locked: consume the click but do nothing
                                return
                            end
                            if ctx.selMat == matEntry.mat then
                                -- Deselect: reset to engine default
                                ctx.selMat = ""
                                armoryMats[ctx.wepName] = ""
                                clStats.armoryMats[ctx.wepName] = nil
                                resetWeaponMaterial(ctx.wepName)
                            else
                                -- Select: apply material
                                ctx.selMat = matEntry.mat
                                armoryMats[ctx.wepName] = matEntry.mat
                                if clStats.armoryFavorites[ctx.wepName] then
                                    clStats.armoryMats[ctx.wepName] = matEntry.mat
                                end
                                applyWeaponMaterial(ctx.wepName, matEntry.mat)
                            end
                            UI.needsUpdate = true
                            queueSaveStats()
                            return
                        end
                    end
                else
                    -- Click outside menu: close it
                    UI.armoryCtx.wepName = nil
                    UI.needsUpdate = true
                end
                return
            end

            -- Armory tab: left-click a weapon row (no context menu open) to toggle favorite
            if UI.tab == TAB_ARMORY and not UI.armoryCtx.wepName then
                for _, item in ipairs(ARMORY_DISPLAY_LIST) do
                    if not item.isHeader then
                        local rowY = g.rowStartY + item.y - UI.scroll
                        if rowY + item.h > g.rowStartY and rowY < g.rowEndY then
                            if hit(g.px, rowY, g.pw - 12, item.h, mx, my) then
                                if clStats.armoryFavorites[item.name] then
                                    -- Unfavourite: remove and prune saved skin
                                    clStats.armoryFavorites[item.name] = nil
                                    clStats.armoryMats[item.name] = nil
                                else
                                    -- Enforce hard limit of 10 favourites
                                    local favCount = 0
                                    for _ in pairs(clStats.armoryFavorites) do favCount = favCount + 1 end
                                    if favCount >= 10 then
                                        -- Silently reject  at limit
                                        return
                                    end
                                    clStats.armoryFavorites[item.name] = true
                                    -- Carry over any existing in-session skin into the saved set
                                    local existingMat = armoryMats[item.name]
                                    if existingMat and existingMat ~= "" then
                                        clStats.armoryMats[item.name] = existingMat
                                    end
                                end
                                rebuildArmoryList()
                                -- Re-clamp scroll since list height may have changed
                                local g2 = getPanelGeom(Cached_HUD_SW, Cached_HUD_SH)
                                UI.scroll = m_min(UI.scroll, m_max(0, ARMORY_TOTAL_H - g2.listAreaH))
                                UI.needsUpdate = true
                                queueSaveStats()
                                return
                            end
                        end
                    end
                end
            end
        end

        -- Mouse2 (right-click): open armory context menu on weapon row
        if k == 108 and UI.open and UI.tab == TAB_ARMORY then
            local mx, my = input.getCursorPos()
            -- Iterate display list to find which weapon row was right-clicked
            for _, item in ipairs(ARMORY_DISPLAY_LIST) do
                if not item.isHeader then
                    local rowY = g.rowStartY + item.y - UI.scroll
                    if rowY + item.h > g.rowStartY and rowY < g.rowEndY then
                        if hit(g.px, rowY, g.pw - 12, item.h, mx, my) then
                            -- Toggle: if the same weapon's menu is already open, close it
                            if UI.armoryCtx.wepName == item.name then
                                UI.armoryCtx.wepName = nil
                            else
                                UI.armoryCtx.wepName = item.name
                                UI.armoryCtx.x       = mx
                                UI.armoryCtx.y       = rowY + item.h
                                UI.armoryCtx.selMat  = armoryMats[item.name] or ""
                            end
                            UI.needsUpdate = true
                            return
                        end
                    end
                end
            end
        end
    end)

    net.receive("TriggerMedal", function(len)
        local medalID = net.readUInt(8)
        local ct = curtime_l()
        local found = false
        
        for i, medal in ipairs(activeMedals) do
            if medal.id == medalID then
                medal.startTime  = ct
                medal.endTime    = ct + CFG.DisplayTime
                medal.multiplier = (medal.multiplier or 1) + 1
                medal.displayStr = CFG.Tiers[medalID + 1].Name .. " x" .. medal.multiplier
                medal.currentX   = nil  -- force snap to new slot position after swap-to-end
                
                local n = #activeMedals
                if i ~= n then
                    activeMedals[i] = activeMedals[n]
                    activeMedals[n] = medal
                end
                found = true
                break
            end
        end
        
        if not found then
            activeMedals[#activeMedals + 1] = { 
                id = medalID, 
                startTime = ct, 
                endTime = ct + CFG.DisplayTime, 
                multiplier = 1,
                displayStr = CFG.Tiers[medalID + 1].Name 
            }
            -- Issue #1 fix: front-pointer compaction instead of table.remove loop.
            -- Avoids O(n) element shifting and GC churn, consistent with the
            -- write-pointer pattern used by lvlNotifs and activeMedals elsewhere.
            if #activeMedals > CFG.Max_Visible then
                local overflow = #activeMedals - CFG.Max_Visible
                for i = 1, CFG.Max_Visible do
                    activeMedals[i] = activeMedals[i + overflow]
                end
                for i = CFG.Max_Visible + 1, CFG.Max_Visible + overflow do
                    activeMedals[i] = nil
                end
            end
        end

        medalStats[medalID] = (medalStats[medalID] or 0) + 1
        updateUICache()
        queueSaveStats()
    end)

    hook.add("DrawHUD", "DrawKillMedal", function()
        if not UI.open and #activeMedals == 0 and #lvlNotifs == 0 and #ObjCompletions == 0 and not ActiveDeathCard then return end

        local ct, ft = curtime_l(), timer.frametime()

        -- Update HUD resolution cache once per frame. scrollDragThink and UIInput
        -- read these cached values instead of calling render.getGameResolution() again.
        local sw, sh = render.getGameResolution()
        Cached_HUD_SW, Cached_HUD_SH = sw, sh

        -- Write-pointer compaction: no table.remove, no element shifting, no GC churn.
        -- Compact lvlNotifs and draw in one combined pass.
        local liveCount = 0
        for i = 1, #lvlNotifs do
            local notif = lvlNotifs[i]
            if ct - notif.startTime <= CFG.Leveling.Notification_Time then
                liveCount = liveCount + 1
                lvlNotifs[liveCount] = notif
            end
        end
        for i = liveCount + 1, #lvlNotifs do lvlNotifs[i] = nil end

        for i = 1, liveCount do
            local notif = lvlNotifs[i]
            local timeSince = ct - notif.startTime
            local alpha = 255
            if timeSince > (CFG.Leveling.Notification_Time - CFG.FadeTime) then
                alpha = 255 * ((CFG.Leveling.Notification_Time - timeSince) / CFG.FadeTime)
            end
            render.setFont(levelFont)
            render.setRGBA(255, 215, 0, m_floor(alpha))
            render.drawSimpleText(sw / 2, (sh / 2) + CFG.Leveling.Notification_Y - (i * 30), notif.displayStr, 1, 1)
        end

        -- Objective completion notifications: slide in from the right edge,
        -- rest at the right-center of the screen, then slide back out right.
        local objCompCount = 0
        for i = 1, #ObjCompletions do
            if ct <= ObjCompletions[i].endTime then
                objCompCount = objCompCount + 1
                ObjCompletions[objCompCount] = ObjCompletions[i]
            end
        end
        for i = objCompCount + 1, #ObjCompletions do ObjCompletions[i] = nil end

        if objCompCount > 0 then
            -- Card dimensions
            local cardW  = 320
            local cardH  = 58
            local cardPad = 12
            local restX  = sw - cardW - 16          -- resting X (just inside right edge)
            local hiddenX = sw + cardW               -- fully off-screen right

            for i = 1, objCompCount do
                local notif        = ObjCompletions[i]
                local timeSince    = ct - notif.startTime
                local timeUntilEnd = notif.endTime - ct
                local alpha = 255
                local slideX

                -- Slide in from right during fade-in window
                if timeSince < CFG.Obj_Notify_Fade then
                    local frac = timeSince / CFG.Obj_Notify_Fade
                    -- ease-out: frac^0.4 gives a quick settle
                    local ease = frac ^ 0.4
                    slideX = hiddenX + (restX - hiddenX) * ease
                    alpha  = 255 * frac
                -- Slide back out right during fade-out window
                elseif timeUntilEnd < CFG.Obj_Notify_Fade then
                    local frac = timeUntilEnd / CFG.Obj_Notify_Fade
                    slideX = restX + (hiddenX - restX) * (1 - frac)
                    alpha  = 255 * frac
                else
                    slideX = restX
                end

                -- Stack cards upward from vertical center-right
                local baseY  = (sh / 2) - (cardH / 2)
                local drawY  = baseY - (i - 1) * (cardH + 6)
                local af     = m_floor(alpha)

                -- Card background
                render.setRGBA(14, 14, 14, m_floor(af * 0.92))
                r_rect(m_floor(slideX), drawY, cardW, cardH)

                -- Left accent stripe (gold)
                render.setRGBA(255, 215, 0, af)
                r_rect(m_floor(slideX), drawY, 3, cardH)

                -- Title text (objective type label)
                render.setFont(objNotifyTitleFont)
                render.setRGBA(255, 215, 0, af)
                render.drawSimpleText(m_floor(slideX) + cardPad + 6, drawY + 14, notif.title, 0, 0)

                -- Objective name
                render.setFont(objNotifySubFont)
                render.setRGBA(220, 220, 220, af)
                render.drawSimpleText(m_floor(slideX) + cardPad + 6, drawY + 36, notif.name, 0, 0)
            end
        end

        -- Write-pointer compaction for activeMedals (preserves display order).
        local medalCount = 0
        for i = 1, #activeMedals do
            if ct <= activeMedals[i].endTime then
                medalCount = medalCount + 1
                activeMedals[medalCount] = activeMedals[i]
            end
        end
        for i = medalCount + 1, #activeMedals do activeMedals[i] = nil end

        -- If the panel was opened this frame (or the frame immediately after),
        -- flush currentX on all live medals including any that arrived via net
        -- after openPanel() was called in the same tick.  This guarantees medals
        -- snap to their correct target position rather than lerping in from stale
        -- coordinates regardless of when in the frame they were added.
        if UI.justOpened then
            for i = 1, medalCount do activeMedals[i].currentX = nil end
            UI.justOpened = false
        end

        if medalCount > 0 then
            local w, h = CFG.Draw_Size_W, CFG.Draw_Size_H
            local totalWidth = (medalCount * w) + ((medalCount - 1) * CFG.Medal_Spacing)
            local startX = (sw / 2) - (totalWidth / 2)
            local wSpacing = w + CFG.Medal_Spacing
            local animTime = CFG.AnimTime
            local fadeTime = CFG.FadeTime
            local dropOff  = CFG.Drop_Offset
            local textOffY = CFG.Text_Offset_Y
            render.setFont(medalFont)
            
            for i = 1, medalCount do
                local medal = activeMedals[i]
                local tierData = CFG.Tiers[medal.id + 1]
                if tierData then
                    local timeSinceStart = ct - medal.startTime
                    local timeUntilEnd   = medal.endTime - ct
                    local currentY = dropOff
                   
                    if timeSinceStart < animTime then
                        local fraction = timeSinceStart / animTime
                        currentY = -h + (dropOff + h) * (fraction * (2 - fraction))
                    end

                    local alpha = timeUntilEnd < fadeTime and m_floor(255 * (timeUntilEnd / fadeTime)) or 255
                    local targetX = startX + ((i - 1) * wSpacing)
                    
                    if not medal.currentX then medal.currentX = targetX end
                    medal.currentX = medal.currentX + (targetX - medal.currentX) * 15 * ft
                    
                    render.setRGBA(255, 255, 255, alpha)
                    render.setMaterial(streakMat)
                    render.drawTexturedRectUV(medal.currentX, currentY, w, h, tierData.u1, tierData.v1, tierData.u2, tierData.v2)
                    render.drawSimpleText(medal.currentX + (w / 2), currentY + h + textOffY, medal.displayStr, 1, 0)
                end
            end
        end

        ------------------------------------------------------------------------
        -- DEATH CARD RENDER
        -- Phase 1: slides in over 0.5s on death.
        -- Phase 2: holds at rest while player is dead (indefinite).
        -- Phase 3: player respawns  hold for HOLD_AFTER_RESPAWN seconds, then
        --          slide out over 0.5s and clear.
        -- A 1s minimum dead-time guard prevents the respawn check firing on the
        -- same frame the card arrives (race between server net message and client
        -- isAlive() state update).
        ------------------------------------------------------------------------
        if ActiveDeathCard then
            local dc = ActiveDeathCard

            -- Only start watching for respawn after the minimum dead-time window.
            -- This prevents a stale isAlive()=true from dismissing the card instantly.
            if not dc.dismissTime
            and (ct - dc.startTime) >= DC_MIN_DEAD_TIME
            and Owner:isValid() and Owner:isAlive() then
                dc.dismissTime = ct
            end

            local cardW   = 480
            local cardH   = 90
            local hiddenY = sh + cardH
            local restY   = sh - cardH - 40

            local timeSince = ct - dc.startTime
            local alpha  = 255
            local slideY = restY

            if timeSince < DC_SLIDE_TIME then
                local frac = timeSince / DC_SLIDE_TIME
                slideY = hiddenY + (restY - hiddenY) * (frac ^ 0.4)
                alpha  = 255 * frac
            elseif dc.dismissTime then
                local sinceRespawn = ct - dc.dismissTime
                if sinceRespawn >= DC_HOLD_AFTER_SPAWN then
                    local frac = m_min((sinceRespawn - DC_HOLD_AFTER_SPAWN) / DC_SLIDE_TIME, 1)
                    slideY = restY + (hiddenY - restY) * frac
                    alpha  = 255 * (1 - frac)
                    if frac >= 1 then
                        ActiveDeathCard = nil
                    end
                end
            end

            if ActiveDeathCard then
                local af = m_floor(alpha)
                local cx = m_floor((sw - cardW) / 2)
                local cy = m_floor(slideY)

                render.setRGBA(14, 14, 14, m_floor(af * 0.95))
                r_rect(cx, cy, cardW, cardH)

                render.setRGBA(220, 40, 40, af)
                r_rect(cx, cy, cardW, 4)

                render.setRGBA(60, 60, 60, af)
                render.drawRectOutline(cx, cy, cardW, cardH, 1)
                render.setRGBA(10, 10, 10, af)
                render.drawRectOutline(cx - 1, cy - 1, cardW + 2, cardH + 2, 1)

                render.setFont(deathMiniFont)
                render.setRGBA(220, 40, 40, af)
                render.drawSimpleText(cx + 16, cy + 10, "KILLED BY", 0, 0)

                render.setFont(deathNameFont)
                render.setRGBA(245, 245, 245, af)
                render.drawSimpleText(cx + 16, cy + 26, dc.name, 0, 0)

                local wepDisplay = (dc.wep and dc.wep ~= "") and dc.wep or "Unknown"
                render.setFont(deathWepFont)
                render.setRGBA(160, 160, 160, af)
                render.drawSimpleText(cx + 16, cy + 52, wepDisplay, 0, 0)

                local statX   = cx + cardW - 15
                local kdRatio = dc.kills / m_max(1, dc.deaths)

                render.setFont(deathSubFont)
                render.setRGBA(220, 220, 220, af)
                render.drawSimpleText(statX, cy + 14, s_fmt("Streak: %d", dc.streak), 2, 0)

                render.setFont(deathMiniFont)
                render.setRGBA(140, 140, 140, af)
                render.drawSimpleText(statX, cy + 46, s_fmt("K/D Ratio: %.2f", kdRatio), 2, 0)
                render.drawSimpleText(statX, cy + 66, s_fmt("Distance: %.0f HU", dc.dist), 2, 0)
            end
        end

        if UI.open then drawStatsPanel(sw, sh) end
    end)
end
